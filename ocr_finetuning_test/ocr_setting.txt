# 웹툰 데이터에 맞게 파인튜닝을 실행하는 명령어 (여기서부터 안됨)

    --train_data data/lmdb_output/train \
    --valid_data data/lmdb_output/validation \
    --Transformation None \
    --FeatureExtraction VGG \
    --SequenceModeling BiLSTM \
    --Prediction CTC \
    --workers 4 \
    --batch_size 64 \
    --num_iter 10000 \
    --valInterval 1000 \
    --saved_model ./saved_models/korean_g2.pth \
