CREATE TABLE webtoon (
    id INT AUTO_INCREMENT PRIMARY KEY,
    group_id INT,
    title VARCHAR(255),
    company VARCHAR(100),
    language VARCHAR(10),  -- 'kr', 'en' 등
    genre INT COMMENT '장르 코드: 0=액션, 1=로맨스, 2=스릴러, 3=무협',
    url TEXT
);


CREATE TABLE webtoon_group (
    id INT AUTO_INCREMENT PRIMARY KEY,
    group_name VARCHAR(255)  -- 예: '화산귀환 / Return of the Mount Hua Sect'
);


CREATE TABLE episode (
    id INT AUTO_INCREMENT PRIMARY KEY,
    webtoon_id INT,
    episode_number INT,
    lang VARCHAR(10), -- 'KR' or 'EN'
    url TEXT,
    jpg_url TEXT
);

CREATE TABLE cut_image (
    id INT AUTO_INCREMENT PRIMARY KEY,
    webtoon_id INT,         -- 웹툰 ID
    episode_id INT,         -- 회차 ID
    cut_number INT,         -- 컷 순서
    image_path TEXT,        -- 저장된 이미지 경로
    height_px INT           -- 이미지 높이 (px 단위)
);

CREATE TABLE dialogue (
    id INT AUTO_INCREMENT PRIMARY KEY,
    cut_image_id INT,
    content TEXT,
    type VARCHAR(50),
    sequence INT,
    speaker_id INT  -- speaker.id를 참조
);

CREATE TABLE speaker (
    id INT AUTO_INCREMENT PRIMARY KEY,
    webtoon_id INT,              -- 어떤 웹툰의 등장인물인지
    name VARCHAR(255) NOT NULL   -- 등장인물 이름
);
